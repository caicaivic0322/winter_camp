## 19.1 栈（后进先出）
```cpp
#include <stack>
using namespace std;

int main() {
    stack<int> st;
    st.push(1); st.push(2); st.push(3);
    while (!st.empty()) {
        cout << st.top() << " ";
        st.pop();
    }
}
```

---

## 19.2 队列（先进先出）
```cpp
#include <queue>
using namespace std;

int main() {
    queue<int> q;
    q.push(10); q.push(20); q.push(30);
    while (!q.empty()) {
        cout << q.front() << " ";
        q.pop();
    }
}
```

## 19.3 应用：括号匹配
```cpp
bool isValid(const string& s) {
    stack<char> st;
    for (char c : s) {
        if (c=='('||c=='['||c=='{') st.push(c);
        else {
            if (st.empty()) return false;
            char t = st.top(); st.pop();
            if ((t=='('&&c!=')')||(t=='['&&c!=']')||(t=='{'&&c!='}')) return false;
        }
    }
    return st.empty();
}
```

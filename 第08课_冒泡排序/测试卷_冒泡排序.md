## 四、编程题参考答案

### 第1题

```cpp
#include <iostream>
using namespace std;
int main() {
    int n, arr[1010];
    cin >> n;
    for (int i = 0; i < n; i++) cin >> arr[i];
    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - 1 - i; j++)
            if (arr[j] > arr[j + 1]) swap(arr[j], arr[j + 1]);
    for (int i = 0; i < n; i++) cout << arr[i] << " ";
    return 0;
}
```

### 第2题

```cpp
#include <iostream>
using namespace std;
int main() {
    int n, arr[1010], count = 0;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> arr[i];
    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - 1 - i; j++)
            if (arr[j] > arr[j + 1]) { swap(arr[j], arr[j + 1]); count++; }
    cout << count << endl;
    return 0;
}
```

## 三、综合题一：高精度阶乘

**题目**：计算 n! 的精确值。

```cpp
#include <iostream>
#include <string>
using namespace std;

string multiply(string num, int x) {
    if (x == 0) return "0";
    string result = "";
    int carry = 0;
    for (int i = num.length() - 1; i >= 0 || carry; i--) {
        int product = carry;
        if (i >= 0) product += (num[i] - '0') * x;
        result = (char)(product % 10 + '0') + result;
        carry = product / 10;
    }
    return result;
}

string factorial(int n) {
    string result = "1";
    for (int i = 2; i <= n; i++) {
        result = multiply(result, i);
    }
    return result;
}

int main() {
    int n;
    cin >> n;
    cout << factorial(n) << endl;
    return 0;
}
```

## 四、综合题二：排名系统

**题目**：输入 n 个学生的成绩，按成绩排序后输出排名。成绩相同的学生排名相同。

```cpp
#include <iostream>
#include <algorithm>
using namespace std;

struct Student {
    string name;
    int score;
    int rank;
};

bool cmp(Student a, Student b) {
    return a.score > b.score;
}

int main() {
    int n;
    cin >> n;

    Student students[110];
    for (int i = 0; i < n; i++) {
        cin >> students[i].name >> students[i].score;
    }

    sort(students, students + n, cmp);

    // 计算排名
    students[0].rank = 1;
    for (int i = 1; i < n; i++) {
        if (students[i].score == students[i-1].score) {
            students[i].rank = students[i-1].rank;
        } else {
            students[i].rank = i + 1;
        }
    }

    // 输出
    for (int i = 0; i < n; i++) {
        cout << students[i].rank << " "
             << students[i].name << " "
             << students[i].score << endl;
    }

    return 0;
}
```

## 五、综合题三：大数加减

**题目**：实现一个计算器，支持两个大整数的加法和减法。

```cpp
#include <iostream>
#include <string>
using namespace std;

// 比较大小
int compare(string a, string b) {
    if (a.length() != b.length())
        return a.length() > b.length() ? 1 : -1;
    return a.compare(b);
}

// 高精度加法
string add(string a, string b) {
    string c = "";
    int carry = 0;
    int i = a.length() - 1, j = b.length() - 1;
    while (i >= 0 || j >= 0 || carry) {
        int sum = carry;
        if (i >= 0) sum += a[i--] - '0';
        if (j >= 0) sum += b[j--] - '0';
        c = (char)(sum % 10 + '0') + c;
        carry = sum / 10;
    }
    return c;
}

// 高精度减法（保证 a >= b）
string subtract(string a, string b) {
    string c = "";
    int borrow = 0;
    int i = a.length() - 1, j = b.length() - 1;
    while (i >= 0) {
        int diff = (a[i--] - '0') - borrow;
        if (j >= 0) diff -= (b[j--] - '0');
        if (diff < 0) { diff += 10; borrow = 1; }
        else { borrow = 0; }
        c = (char)(diff + '0') + c;
    }
    // 去前导零
    int start = 0;
    while (start < c.length() - 1 && c[start] == '0') start++;
    return c.substr(start);
}

int main() {
    string a, op, b;
    cin >> a >> op >> b;

    if (op == "+") {
        cout << add(a, b) << endl;
    } else {
        int cmp = compare(a, b);
        if (cmp == 0) cout << "0" << endl;
        else if (cmp > 0) cout << subtract(a, b) << endl;
        else cout << "-" << subtract(b, a) << endl;
    }

    return 0;
}
```

## 六、综合题四：区间操作

**题目**：给定 n 个区间，找出所有区间的并集长度。

```cpp
#include <iostream>
#include <algorithm>
using namespace std;

struct Interval {
    int start, end;
};

bool cmp(Interval a, Interval b) {
    return a.start < b.start;
}

int main() {
    int n;
    cin >> n;

    Interval intervals[110];
    for (int i = 0; i < n; i++) {
        cin >> intervals[i].start >> intervals[i].end;
    }

    sort(intervals, intervals + n, cmp);

    int totalLength = 0;
    int curStart = intervals[0].start;
    int curEnd = intervals[0].end;

    for (int i = 1; i < n; i++) {
        if (intervals[i].start <= curEnd) {
            curEnd = max(curEnd, intervals[i].end);
        } else {
            totalLength += curEnd - curStart;
            curStart = intervals[i].start;
            curEnd = intervals[i].end;
        }
    }
    totalLength += curEnd - curStart;

    cout << totalLength << endl;
    return 0;
}
```

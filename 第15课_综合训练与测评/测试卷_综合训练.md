### 第1题

```cpp
#include <iostream>
#include <iomanip>
using namespace std;

int main() {
    int n;
    cin >> n;

    string names[110];
    int scores[110];
    int maxS = -1, minS = 101, sum = 0, pass = 0;

    for (int i = 0; i < n; i++) {
        cin >> names[i] >> scores[i];
        maxS = max(maxS, scores[i]);
        minS = min(minS, scores[i]);
        sum += scores[i];
        if (scores[i] >= 60) pass++;
    }

    cout << "最高分: " << maxS << endl;
    cout << "最低分: " << minS << endl;
    cout << fixed << setprecision(2);
    cout << "平均分: " << (double)sum / n << endl;
    cout << "及格率: " << (double)pass * 100 / n << "%" << endl;

    return 0;
}
```

### 第2题

```cpp
#include <iostream>
#include <string>
using namespace std;

string add(string a, string b) {
    string c = "";
    int carry = 0;
    int i = a.length() - 1, j = b.length() - 1;
    while (i >= 0 || j >= 0 || carry) {
        int sum = carry;
        if (i >= 0) sum += a[i--] - '0';
        if (j >= 0) sum += b[j--] - '0';
        c = (char)(sum % 10 + '0') + c;
        carry = sum / 10;
    }
    return c;
}

int main() {
    int n;
    cin >> n;
    string result = "0";
    for (int i = 0; i < n; i++) {
        string num;
        cin >> num;
        result = add(result, num);
    }
    cout << result << endl;
    return 0;
}
```

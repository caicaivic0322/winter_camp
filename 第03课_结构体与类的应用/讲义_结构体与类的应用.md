## 二、结构体回顾

### 2.1 为什么需要结构体？

**问题场景**：管理学生信息（学号、姓名、成绩）

```cpp
// 不好的方式：多个数组
int ids[100];
string names[100];
int scores[100];
// 问题：数据分散，不好管理

// 好的方式：结构体
struct Student {
    int id;
    string name;
    int score;
};
Student students[100];  // 一个数组搞定！
```

### 2.2 结构体基础语法

```cpp
// 定义结构体
struct Point {
    int x;
    int y;
};

// 使用结构体
Point p1;           // 声明
p1.x = 10;          // 访问成员
p1.y = 20;

Point p2 = {30, 40};  // 初始化
```

## 三、类的基本概念

### 3.1 从结构体到类

**类 = 数据 + 操作数据的函数**

```cpp
// 结构体版本
struct Rectangle {
    int width;
    int height;
};

int getArea(Rectangle r) {
    return r.width * r.height;
}

// 类版本：把函数放进来
class Rectangle {
public:
    int width;
    int height;

    int getArea() {
        return width * height;
    }
};
```

### 3.2 类的基本结构

```cpp
class 类名 {
public:      // 公有成员：外部可访问
    // 成员变量
    // 成员函数

private:     // 私有成员：仅类内部可访问
    // 成员变量
    // 成员函数
};
```

### 3.3 构造函数

**构造函数**：创建对象时自动调用，用于初始化

```cpp
class Student {
public:
    string name;
    int score;

    // 构造函数（与类名相同，无返回值）
    Student() {
        name = "";
        score = 0;
    }

    // 带参数的构造函数
    Student(string n, int s) {
        name = n;
        score = s;
    }

    void print() {
        cout << name << ": " << score << endl;
    }
};

int main() {
    Student s1;                    // 调用无参构造
    Student s2("张三", 90);        // 调用带参构造
    s2.print();                    // 输出: 张三: 90
    return 0;
}
```

### 3.4 struct vs class

| 特性         | struct       | class        |
| ------------ | ------------ | ------------ |
| 默认访问权限 | public       | private      |
| 常用场景     | 简单数据组合 | 复杂对象封装 |
| C 兼容性     | 兼容         | 不兼容       |

**实际建议**：简单数据用 struct，需要封装时用 class

## 四、📝 实战项目一：学生成绩管理系统

### 题目描述

设计一个学生成绩管理系统，实现以下功能：

1. 录入学生信息
2. 按成绩排序
3. 查找学生
4. 统计及格率

### 代码实现

```cpp
#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

class Student {
public:
    string name;
    int score;

    Student() : name(""), score(0) {}
    Student(string n, int s) : name(n), score(s) {}

    void input() {
        cin >> name >> score;
    }

    void print() {
        cout << name << " " << score << endl;
    }

    bool isPassed() {
        return score >= 60;
    }
};

class ScoreSystem {
private:
    Student students[100];
    int count;

public:
    ScoreSystem() : count(0) {}

    // 添加学生
    void addStudent(string name, int score) {
        students[count++] = Student(name, score);
    }

    // 输入多个学生
    void inputStudents(int n) {
        count = n;
        for (int i = 0; i < n; i++) {
            students[i].input();
        }
    }

    // 按成绩降序排序
    void sortByScore() {
        for (int i = 0; i < count - 1; i++) {
            for (int j = 0; j < count - 1 - i; j++) {
                if (students[j].score < students[j + 1].score) {
                    swap(students[j], students[j + 1]);
                }
            }
        }
    }

    // 查找学生
    Student* findStudent(string name) {
        for (int i = 0; i < count; i++) {
            if (students[i].name == name) {
                return &students[i];
            }
        }
        return nullptr;
    }

    // 计算及格率
    double getPassRate() {
        int passed = 0;
        for (int i = 0; i < count; i++) {
            if (students[i].isPassed()) passed++;
        }
        return count > 0 ? (double)passed / count * 100 : 0;
    }

    // 打印所有学生
    void printAll() {
        for (int i = 0; i < count; i++) {
            students[i].print();
        }
    }

    // 获取最高分
    Student getTop() {
        int maxIdx = 0;
        for (int i = 1; i < count; i++) {
            if (students[i].score > students[maxIdx].score) {
                maxIdx = i;
            }
        }
        return students[maxIdx];
    }
};

int main() {
    ScoreSystem sys;

    cout << "请输入学生人数: ";
    int n;
    cin >> n;

    cout << "请输入每个学生的姓名和成绩:" << endl;
    sys.inputStudents(n);

    cout << "\n按成绩排序后:" << endl;
    sys.sortByScore();
    sys.printAll();

    cout << "\n及格率: " << sys.getPassRate() << "%" << endl;
    cout << "最高分: ";
    sys.getTop().print();

    return 0;
}
```

## 五、📝 实战项目二：银行账户系统

### 题目描述

设计银行账户类，支持：存款、取款、查询余额、转账

### 代码实现

```cpp
#include <iostream>
#include <string>
using namespace std;

class BankAccount {
private:
    string owner;      // 户主
    double balance;    // 余额
    string accountId;  // 账号

public:
    // 构造函数
    BankAccount(string id, string name, double initial = 0) {
        accountId = id;
        owner = name;
        balance = initial;
    }

    // 存款
    void deposit(double amount) {
        if (amount > 0) {
            balance += amount;
            cout << "存款成功! 当前余额: " << balance << endl;
        } else {
            cout << "存款金额必须大于0!" << endl;
        }
    }

    // 取款
    bool withdraw(double amount) {
        if (amount > balance) {
            cout << "余额不足!" << endl;
            return false;
        }
        if (amount <= 0) {
            cout << "取款金额必须大于0!" << endl;
            return false;
        }
        balance -= amount;
        cout << "取款成功! 当前余额: " << balance << endl;
        return true;
    }

    // 查询余额
    double getBalance() {
        return balance;
    }

    // 转账
    bool transfer(BankAccount &target, double amount) {
        if (withdraw(amount)) {
            target.deposit(amount);
            cout << "转账成功!" << endl;
            return true;
        }
        return false;
    }

    // 显示账户信息
    void showInfo() {
        cout << "账号: " << accountId << endl;
        cout << "户主: " << owner << endl;
        cout << "余额: " << balance << endl;
    }
};

int main() {
    BankAccount acc1("001", "张三", 1000);
    BankAccount acc2("002", "李四", 500);

    cout << "=== 初始状态 ===" << endl;
    acc1.showInfo();
    cout << endl;
    acc2.showInfo();

    cout << "\n=== 张三存款200 ===" << endl;
    acc1.deposit(200);

    cout << "\n=== 张三取款500 ===" << endl;
    acc1.withdraw(500);

    cout << "\n=== 张三转账300给李四 ===" << endl;
    acc1.transfer(acc2, 300);

    cout << "\n=== 最终状态 ===" << endl;
    acc1.showInfo();
    cout << endl;
    acc2.showInfo();

    return 0;
}
```

## 六、数据结构入门：栈与队列 (Stack & Queue)

在学习了类和对象之后，我们可以利用它们来封装一些经典的数据结构。

### 6.1 栈 (Stack) - "后进先出"

**概念**：栈就像一个羽毛球筒，最先放进去的球最后才能拿出来。
**特点**：后进先出 (LIFO - Last In First Out)
**常用操作**：

- `push`: 入栈（放入顶部）
- `pop`: 出栈（移除顶部）
- `top`: 查看顶部元素
- `empty`: 判断是否为空

**实战任务**：使用 `vector` 实现一个整数栈 `MyStack`

```cpp
#include <iostream>
#include <vector>
using namespace std;

class MyStack {
private:
    vector<int> data;

public:
    // 入栈
    void push(int val) {
        data.push_back(val);
    }

    // 出栈
    void pop() {
        if (!empty()) {
            data.pop_back();
        }
    }

    // 获取栈顶元素
    int top() {
        if (!empty()) {
            return data.back();
        }
        throw runtime_error("Stack is empty");
    }

    // 判断为空
    bool empty() {
        return data.empty();
    }

    // 获取大小
    int size() {
        return data.size();
    }
};

int main() {
    MyStack s;
    s.push(10);
    s.push(20);
    s.push(30);

    cout << "栈顶元素: " << s.top() << endl; // 30
    s.pop();
    cout << "弹出后栈顶: " << s.top() << endl; // 20

    return 0;
}
```

### 6.2 队列 (Queue) - "先进先出"

**概念**：队列就像排队买票，先来的人先买。
**特点**：先进先出 (FIFO - First In First Out)
**常用操作**：

- `push`: 入队（放入队尾）
- `pop`: 出队（移除队头）
- `front`: 查看队头元素
- `empty`: 判断是否为空

**实战任务**：使用两个 `vector` 实现一个队列 `MyQueue`

**思考**：为什么要用两个 vector？

- `vector` 在头部删除元素 (`erase(begin())`) 效率较低，因为后面的元素都要向前移动。
- 我们可以用两个栈（`inStack` 和 `outStack`）来模拟队列，保证所有操作依然高效。

**实现逻辑**：

1. **入队**：直接 push 进入 `inStack`。
2. **出队/查看队头**：
   - 如果 `outStack` 不为空，直接从 `outStack` 顶部取。
   - 如果 `outStack` 为空，将 `inStack` 所有元素倒序倒入 `outStack`，然后再取。

```cpp
#include <iostream>
#include <vector>
using namespace std;

class MyQueue {
private:
    vector<int> inStack;   // 负责进
    vector<int> outStack;  // 负责出

    // 辅助函数：将 inStack 数据转移到 outStack
    void move() {
        if (outStack.empty()) {
            while (!inStack.empty()) {
                outStack.push_back(inStack.back());
                inStack.pop_back();
            }
        }
    }

public:
    // 入队
    void push(int x) {
        inStack.push_back(x);
    }

    // 出队
    void pop() {
        move();
        if (!outStack.empty()) {
            outStack.pop_back();
        }
    }

    // 获取队头元素
    int front() {
        move();
        if (!outStack.empty()) {
            return outStack.back();
        }
        throw runtime_error("Queue is empty");
    }

    // 判断为空
    bool empty() {
        return inStack.empty() && outStack.empty();
    }
};

int main() {
    MyQueue q;
    q.push(1);
    q.push(2);
    q.push(3);

    cout << "队头元素: " << q.front() << endl; // 1
    q.pop();
    cout << "出队一次后队头: " << q.front() << endl; // 2

    q.push(4);
    cout << "加入4后队头: " << q.front() << endl; // 2 (依然是2)

    return 0;
}
```

## 七、本课小结

### 设计思想

1. **数据抽象**：将现实事物抽象为结构体/类
2. **封装**：数据和操作放在一起
3. **模块化**：分解复杂问题

### 应用场景对比

| 场景         | 推荐使用         | 原因       |
| ------------ | ---------------- | ---------- |
| 简单数据组合 | struct           | 简洁明了   |
| 需要隐藏数据 | class + private  | 防止误操作 |
| 需要初始化   | class + 构造函数 | 自动初始化 |
| 复杂系统     | class            | 方便扩展   |

### 编程技巧

- ✅ 先设计数据结构，再写操作函数
- ✅ 构造函数做好初始化
- ✅ 成员函数保持简单，一个函数做一件事
- ✅ 用 private 保护关键数据

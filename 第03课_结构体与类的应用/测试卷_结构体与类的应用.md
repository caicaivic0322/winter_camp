### 第1题（15分）：矩形类

**题目描述**：
设计一个矩形类 `Rectangle`，包含：

- 私有成员：宽度 `width`、高度 `height`
- 公有成员函数：
  - 构造函数（带参数）
  - `getArea()` 返回面积
  - `getPerimeter()` 返回周长
  - `print()` 输出矩形信息

**输入格式**：两个整数，分别表示宽度和高度

**输出格式**：

- 第一行：宽度和高度
- 第二行：面积
- 第三行：周长

**输入样例**：

```
5 3
```

**输出样例**：

```
宽度: 5, 高度: 3
面积: 15
周长: 16
```

### 第1题

```cpp
#include <iostream>
using namespace std;

class Rectangle {
private:
    int width, height;
public:
    Rectangle(int w, int h) : width(w), height(h) {}
    int getArea() { return width * height; }
    int getPerimeter() { return 2 * (width + height); }
    void print() {
        cout << "宽度: " << width << ", 高度: " << height << endl;
        cout << "面积: " << getArea() << endl;
        cout << "周长: " << getPerimeter() << endl;
    }
};

int main() {
    int w, h;
    cin >> w >> h;
    Rectangle rect(w, h);
    rect.print();
    return 0;
}
```

### 第2题

```cpp
#include <iostream>
using namespace std;

class Counter {
private:
    int count;
public:
    Counter() : count(0) {}
    void increment() { count++; }
    void decrement() { if (count > 0) count--; }
    void reset() { count = 0; }
    int getValue() { return count; }
};

int main() {
    Counter c;
    char op;
    while (cin >> op) {
        if (op == 'I') c.increment();
        else if (op == 'D') c.decrement();
        else if (op == 'R') c.reset();
    }
    cout << c.getValue() << endl;
    return 0;
}
```

### 第3题

```cpp
#include <iostream>
#include <iomanip>
using namespace std;

class Student {
public:
    string name;
    int score;
};

int main() {
    int n;
    cin >> n;
    Student students[110];
    int sum = 0, maxIdx = 0, minIdx = 0;

    for (int i = 0; i < n; i++) {
        cin >> students[i].name >> students[i].score;
        sum += students[i].score;
        if (students[i].score > students[maxIdx].score) maxIdx = i;
        if (students[i].score < students[minIdx].score) minIdx = i;
    }

    cout << "最高分: " << students[maxIdx].name << " " << students[maxIdx].score << endl;
    cout << "最低分: " << students[minIdx].name << " " << students[minIdx].score << endl;
    cout << fixed << setprecision(2);
    cout << "平均分: " << (double)sum / n << endl;

    return 0;
}
```

### 第4题

```cpp
#include <iostream>
using namespace std;

class BankAccount {
private:
    string owner;
    double balance;
public:
    BankAccount(string name, double init) : owner(name), balance(init) {}

    void deposit(double amount) {
        balance += amount;
        cout << "存款 " << amount << " 成功, 余额: " << balance << endl;
    }

    bool withdraw(double amount) {
        if (amount > balance) {
            cout << "取款 " << amount << " 失败, 余额不足" << endl;
            return false;
        }
        balance -= amount;
        cout << "取款 " << amount << " 成功, 余额: " << balance << endl;
        return true;
    }

    double getBalance() { return balance; }
};

int main() {
    BankAccount acc("张三", 1000);
    acc.deposit(500);
    acc.withdraw(200);
    acc.withdraw(2000);
    acc.deposit(300);
    cout << "最终余额: " << acc.getBalance() << endl;
    return 0;
}
```

### 第1题：分区操作（Lomuto 分区）

```cpp
#include <iostream>
#include <algorithm>
using namespace std;

int partition(int arr[], int low, int high) {
    int pivot = arr[______①];  // 选择最后一个元素作为基准
    int i = low - 1;

    for (int j = low; j < ______②; j++) {
        // 如果当前元素小于等于基准
        if (arr[j] <= pivot) {
            i++;
            swap(arr[i], arr[______③]);
        }
    }

    swap(arr[______④], arr[high]);
    return i + 1;
}
```

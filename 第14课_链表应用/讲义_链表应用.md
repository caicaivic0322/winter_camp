## 二、经典问题一：反转链表

### 2.1 迭代法

```cpp
Node* reverse(Node* head) {
    Node* prev = nullptr;
    Node* cur = head;

    while (cur != nullptr) {
        Node* next = cur->next;  // 保存下一个
        cur->next = prev;        // 反转指针
        prev = cur;              // 移动 prev
        cur = next;              // 移动 cur
    }

    return prev;
}
```

### 2.2 图示

```
原始: 1 -> 2 -> 3 -> NULL

步骤1: NULL <- 1    2 -> 3 -> NULL
步骤2: NULL <- 1 <- 2    3 -> NULL
步骤3: NULL <- 1 <- 2 <- 3

结果: 3 -> 2 -> 1 -> NULL
```

## 三、经典问题二：合并两个有序链表

```cpp
Node* mergeTwoLists(Node* l1, Node* l2) {
    Node dummy(0);  // 哨兵节点
    Node* tail = &dummy;

    while (l1 && l2) {
        if (l1->data <= l2->data) {
            tail->next = l1;
            l1 = l1->next;
        } else {
            tail->next = l2;
            l2 = l2->next;
        }
        tail = tail->next;
    }

    tail->next = l1 ? l1 : l2;
    return dummy.next;
}
```

## 四、经典问题三：删除倒数第N个节点

### 快慢指针法

```cpp
Node* removeNthFromEnd(Node* head, int n) {
    Node dummy(0);
    dummy.next = head;
    Node* fast = &dummy;
    Node* slow = &dummy;

    // fast 先走 n+1 步
    for (int i = 0; i <= n; i++) {
        fast = fast->next;
    }

    // 同时移动
    while (fast) {
        fast = fast->next;
        slow = slow->next;
    }

    // slow->next 就是要删除的节点
    Node* temp = slow->next;
    slow->next = slow->next->next;
    delete temp;

    return dummy.next;
}
```

## 五、经典问题四：判断链表是否有环

### 快慢指针法

```cpp
bool hasCycle(Node* head) {
    Node* slow = head;
    Node* fast = head;

    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) return true;
    }

    return false;
}
```

### 找环的入口

```cpp
Node* detectCycle(Node* head) {
    Node* slow = head;
    Node* fast = head;

    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;

        if (slow == fast) {
            Node* ptr = head;
            while (ptr != slow) {
                ptr = ptr->next;
                slow = slow->next;
            }
            return ptr;
        }
    }

    return nullptr;
}
```

## 六、经典问题五：找链表中点

```cpp
Node* findMiddle(Node* head) {
    Node* slow = head;
    Node* fast = head;

    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
    }

    return slow;
}
```

## 七、静态链表（数组模拟）

竞赛中常用数组模拟链表，避免动态内存：

```cpp
const int MAXN = 100010;
int head, idx;      // head为头指针，idx为当前可用位置
int e[MAXN];        // 数据
int ne[MAXN];       // 下一个节点

void init() {
    head = -1;
    idx = 0;
}

void insertAtHead(int x) {
    e[idx] = x;
    ne[idx] = head;
    head = idx++;
}

void deleteAfter(int k) {
    ne[k] = ne[ne[k]];
}

void traverse() {
    for (int i = head; i != -1; i = ne[i]) {
        cout << e[i] << " ";
    }
}
```
